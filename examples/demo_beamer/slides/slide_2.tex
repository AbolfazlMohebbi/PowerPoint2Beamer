\begin{frame}{Feedback control}
\begin{itemize}
  \item A controller compares the \textbf{output} with the \textit{reference}
  \begin{itemize}
    \item Proportional, integral and derivative actions
    \item Tuned from a model or from experiments
  \end{itemize}
  \item Too much gain makes the loop \alert{\textbf{unstable!}}
  \item A well-tuned loop is \textcolor[RGB]{0,176,80}{fast and stable}
\end{itemize}
\end{frame}
